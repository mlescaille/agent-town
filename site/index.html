Quickstart

>_ git clone https://github.com/mlescaille/agent-town.git

>_ cd agent-town && pnpm i && pnpm build

>_ pnpm exec agentlicense keys init

>_ pnpm exec agentlicense issue --user alice --agent planner --mandate user_goal --goal "Summarize the primary calendar" --actions calendar.read,calendar.create --locations calendar:primary --budget 10 --ttl 3600 > license.jwt

>_ pnpm exec agentlicense check --license "$(cat license.jwt)" --action calendar.read --location calendar:primary --why "Reading today's events is how the summary gets its facts."
